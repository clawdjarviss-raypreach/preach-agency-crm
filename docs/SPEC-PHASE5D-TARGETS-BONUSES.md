## Executive Summary

Phase 5D implements a comprehensive bonus calculation engine that automates Rayan's weekly bonus workflow. This replaces manual spreadsheet calculations with real-time target tracking, automatic bonus detection, and a streamlined approval/payout flow.

**Business Value:**
- Eliminates ~2h/week of manual bonus calculations
- Provides chatters visibility into their bonus progress (motivation)
- Creates auditable bonus history for payroll
- Handles complex rules (creator-specific targets, holiday multipliers, tiered shift bonuses)

```typescript
// convex/schema.ts additions

// Weekly targets per creator (set by admin)
crm_weekly_targets: defineTable({
  creatorId: v.id("crm_creators"),
  weekStart: v.string(), // "2026-02-03" (Monday)
  responseTimeTarget: v.number(), // seconds (e.g., 90 for 1:30)
  ppvUnlockTarget: v.number(), // percentage (e.g., 70)
  ppvMinSent: v.number(), // count (e.g., 75 or 20)
  weeklyBonusAmount: v.number(), // USD cents (e.g., 5000 for $50)
  isActive: v.boolean(),
})
  .index("by_creator_week", ["creatorId", "weekStart"])
  .index("by_week", ["weekStart"]),

// Per-chatter target progress (computed daily)
crm_target_progress: defineTable({
  chatterId: v.id("crm_chatters"),
  creatorId: v.id("crm_creators"),
  weekStart: v.string(),
  avgResponseTime: v.optional(v.number()), // seconds
  ppvUnlockRate: v.optional(v.number()), // percentage
  ppvSentCount: v.optional(v.number()), // count
  meetsResponseTime: v.boolean(),
  meetsPpvUnlock: v.boolean(),
  meetsPpvSent: v.boolean(),
  meetsAllTargets: v.boolean(),
  lastUpdated: v.number(),
})
  .index("by_chatter_week", ["chatterId", "weekStart"])
  .index("by_chatter_creator_week", ["chatterId", "creatorId", "weekStart"]),

// Bonus records (computed + manual)
crm_bonuses: defineTable({
  chatterId: v.id("crm_chatters"),
  type: v.union(
    v.literal("weekly_target"), // Meets all 3 targets for a creator
    v.literal("shift_500"), // $500+ shift 3x in period
    v.literal("shift_750"), // $750+ shift 3x in period
    v.literal("holiday"), // Holiday multiplier bonus
    v.literal("commission"), // 3%/6% sales commission
    v.literal("ad_hoc"), // Manual entry (Christmas, performance, etc.)
  ),
  amount: v.number(), // USD cents
  periodStart: v.string(), // Reference period start
  periodEnd: v.string(), // Reference period end
  creatorId: v.optional(v.id("crm_creators")), // For target/shift bonuses
  description: v.string(),
  status: v.union(
    v.literal("pending"), // Computed, awaiting approval
    v.literal("approved"), // Approved by admin
    v.literal("rejected"), // Rejected
    v.literal("paid"), // Marked as paid
  ),
  approvedBy: v.optional(v.id("crm_chatters")), // Admin who approved
  approvedAt: v.optional(v.number()),
  paidAt: v.optional(v.number()),
  paymentMethod: v.optional(v.string()), // "USDC", "USDT", "Bank"
  paymentRef: v.optional(v.string()), // Transaction ID
  createdAt: v.number(),
  autoGenerated: v.boolean(), // true if computed by system
})
  .index("by_chatter", ["chatterId"])
  .index("by_chatter_period", ["chatterId", "periodStart"])
  .index("by_status", ["status"])
  .index("by_type_period", ["type", "periodStart"]),

// Holiday configuration
crm_holidays: defineTable({
  date: v.string(), // "2026-12-25"
  name: v.string(), // "Christmas"
  hourlyMultiplier: v.number(), // 2.0
  bonusMultiplier: v.number(), // 2.0
  commissionRate: v.number(), // 0.06 (6%)
  isActive: v.boolean(),
})
  .index("by_date", ["date"]),

// Shift bonus tracking (intermediate computation)
crm_shift_bonus_tracker: defineTable({
  chatterId: v.id("crm_chatters"),
  periodStart: v.string(), // Bi-weekly or weekly
  highShiftCount500: v.number(), // Count of $500+ shifts
  highShiftCount750: v.number(), // Count of $750+ shifts
  qualifiesFor500Bonus: v.boolean(),
  qualifiesFor750Bonus: v.boolean(),
  lastUpdated: v.number(),
})
  .index("by_chatter_period", ["chatterId", "periodStart"]),
```
